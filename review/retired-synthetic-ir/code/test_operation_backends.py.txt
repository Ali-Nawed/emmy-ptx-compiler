#!/usr/bin/env python3
"""Explicitly hand-typed backend tests, separate from PyTorch-generated tests.

Exercise direct math nodes, broad finite ranges, special-value classifications,
and reduction combiners the pinned Torch tracer cannot currently produce.
All uploaded PTX comes from our compile.py; never from hand-written PTX/CUDA.
"""

import argparse
import copy
from dataclasses import replace
import hashlib
import json
import math
from pathlib import Path
import subprocess
import sys
import time

import numpy as np

from test_extended import ROOT, check_runner, compare_outputs
from ir_reference import calculate, simulate
from compile import emit_ptx, load_kernel
from submit import RUNNER, post
from emmy.compiler.dim import Dim
from emmy.compiler.dtype import F16, F32, F64, I32, I64
from emmy.compiler.ir.elementwise import ElementwiseImpl
from emmy.compiler.ir.expr import Literal, Var
from emmy.compiler.ir.kernel.ir import Smem, TreeHalve, WarpShuffle
from emmy.compiler.ir.stmt import Accum, Assign, Let, Load, StridedLoop, Write

TYPES = {dtype.name: dtype for dtype in (F16, F32, F64, I32, I64)}
NP_TYPES = {name: np.dtype(name.replace("f", "float") if name.startswith("f") else name.replace("i", "int")) for name in TYPES}


def assignment(name, operation, args, dtype):
    return Assign(name=name, op=ElementwiseImpl(operation), args=tuple(args), dtype=TYPES[dtype])


def save_source(folder, source, meta, expected, expected_filename="expected.npz"):
    folder.mkdir(parents=True, exist_ok=True)
    graph = copy.deepcopy(source.graph)
    graph.nodes[source.node_name] = replace(graph.nodes[source.node_name], op=source.kernel)
    payload = graph.to_dict()
    arguments = {arg["name"]: arg for descriptor in meta["kernels"] for arg in descriptor["args"]}
    for node in payload["nodes"].values():
        for tensor in node.get("outputs", []):
            if tensor["name"] in arguments:
                arg = arguments[tensor["name"]]
                tensor["shape"], tensor["dtype"] = arg["shape"], arg["dtype"]
    # Match Emmy's own dump serializer for its statement repr payloads.
    (folder / f"{source.kernel.name}.ir.json").write_text(json.dumps(payload, indent=2, default=str)+"\n")
    (folder / "meta.json").write_text(json.dumps(meta, indent=2)+"\n")
    np.savez(folder / expected_filename, **expected)
    if expected_filename != "expected.npz":
        # Classification fixtures use their custom checker. Do not leave an
        # ordinary expected.npz that falsely describes the encoded word output.
        (folder / "expected.npz").unlink(missing_ok=True)
    (folder / "provenance.json").write_text(json.dumps({
        "kind": "hand-typed backend fixture, NOT freshly generated by PyTorch",
        "reason": "direct instruction/range/combiner coverage independent of upstream tracer limitations"}, indent=2)+"\n")


def vector_case(dtype, operation, x, y=None, classify=False):
    x = np.asarray(x, dtype=NP_TYPES[dtype])
    y = np.asarray(y if y is not None else np.ones_like(x), dtype=x.dtype)
    source = load_kernel(next((ROOT / f"tests/extended/ext_add_{dtype}_129").glob("*.ir.json")))
    meta = copy.deepcopy(source.metadata)
    xi, yi, output = meta["arg_order"]
    tile = source.kernel.body[0]
    axis = replace(tile.axes[0], extent=Dim(len(x)))
    body = [Load(names=("a",), input=xi, index=(Var(axis.name),), dtype=TYPES[dtype]),
            Load(names=("b",), input=yi, index=(Var(axis.name),), dtype=TYPES[dtype])]
    operands = [x, y] if operation in ("power", "maximum", "minimum") else [x]
    args = ("a", "b") if len(operands) == 2 else ("a",)
    if operation in ("maximum_nan", "minimum_nan"):
        body.append(assignment("root", "sqrt", ("a",), dtype))
        operation = operation.split("_")[0]
        operands, args = [calculate("sqrt", [x], dtype), y], ("root", "b")
    body.append(assignment("answer", operation, args, dtype))
    expected = calculate(operation, operands, dtype)
    stored, output_dtype = "answer", dtype
    if classify:
        if dtype == "f16":
            body.append(assignment("wide", "copy", (stored,), "f32"))
            stored = "wide"
        bit_type = "i64" if dtype == "f64" else "i32"
        body.append(assignment("bits", "bitcast", (stored,), bit_type))
        stored = "bits"
        if dtype == "f64":
            body += [Let(name="shift", value=Literal(32, "int"), dtype=I64),
                     assignment("high", "right_shift", (stored, "shift"), "i64")]
            stored = "high"
        body.append(assignment("encoded", "copy", (stored,), "i32"))
        stored, output_dtype = "encoded", "i32"
    body.append(Write(output=output, index=(Var(axis.name),), values=(stored,), value_dtype=TYPES[output_dtype]))
    source = replace(source, kernel=replace(source.kernel, body=(replace(tile, axes=(axis,), body=tuple(body)),)))
    meta.update(grid=[math.ceil(len(x)/256), 1, 1], block=[256, 1, 1])
    for arg in meta["args"]:
        arg["shape"] = [len(x)]
        if arg["name"] == output:
            arg["dtype"] = output_dtype
        else:
            arg.pop("seed", None)
            arg["data"] = (x if arg["name"] == xi else y).tolist()
    full_meta = {"id": "typed_backend", "kernels": [meta], "graph_inputs": [xi, yi], "graph_outputs": [output],
                 "rtol": {"f16": 2e-3, "f32": 2e-6, "f64": 4e-13, "i32": 0, "i64": 0}[dtype], "atol": 0}
    return source, full_meta, {output: expected}


def reduction_case(dtype, operation, work):
    folder = ROOT / f"tests/extended/ext_row_sum_{dtype}_3x129"
    source = load_kernel(next(folder.glob("*.ir.json")))
    meta = copy.deepcopy(source.metadata)
    data_type = TYPES[dtype]
    tile = source.kernel.body[0]
    co = tile.axes[-1].name

    def rewrite(stmt):
        nested = stmt.nested()
        if nested:
            stmt = stmt.with_bodies(tuple(tuple(rewrite(child) for child in body) for body in nested))
        if isinstance(stmt, Accum):
            return replace(stmt, dtype=data_type, op=ElementwiseImpl(operation))
        if isinstance(stmt, Assign):
            return replace(stmt, dtype=data_type,
                           op=ElementwiseImpl(operation) if stmt.op.name == "add" else stmt.op)
        if isinstance(stmt, Load):
            return replace(stmt, dtype=data_type)
        if isinstance(stmt, Smem):
            return replace(stmt, dtype={"f16": "__half", "f32": "float", "f64": "double", "i32": "int", "i64": "long long"}[dtype], extents=(work//32,))
        if isinstance(stmt, WarpShuffle):
            return replace(stmt, dtype=data_type, combine_states=tuple(rewrite(s) for s in stmt.combine_states))
        if isinstance(stmt, TreeHalve):
            return replace(stmt, dtype=data_type, length=work//32,
                           combine_states=tuple(rewrite(s) for s in stmt.combine_states))
        if isinstance(stmt, StridedLoop):
            return replace(stmt, step=Literal(work, "int"))
        if isinstance(stmt, Write) and dtype == "i64":
            # Keep returned values small enough for the runner's JSON path.
            return stmt
        return stmt

    axes = tuple(replace(axis, extent=Dim(work)) if axis.name == co else axis for axis in tile.axes)
    body = [rewrite(stmt) for stmt in tile.body]
    values = np.resize(np.array([-7, -2, 3, 5, -1], dtype=NP_TYPES[dtype]), (3, 129))
    if operation == "multiply":
        values.fill(1)
        values[:, 0], values[:, 64], values[:, 128] = 2, -1, 2
    # For wide extrema, construct a large exactly represented integer on GPU
    # from small inputs. This avoids input/output JSON numeric precision issues.
    if dtype == "i64" and operation != "multiply":
        def wide_load(stmt):
            if isinstance(stmt, Load) and stmt.input == "x":
                extra = [stmt, Let(name="wide_base", value=Literal(2**60, "int"), dtype=I64)]
                extra += [assignment(name, "add", (name, "wide_base"), "i64") for name in stmt.names]
                return extra
            nested = stmt.nested()
            if nested:
                stmt = stmt.with_bodies(tuple(tuple(result for child in children for result in wide_load(child)) for children in nested))
            return [stmt]
        body = [result for stmt in body for result in wide_load(stmt)]
        expression_values = values + np.int64(2**60)
    else:
        expression_values = values
    functions = {"maximum": np.max, "minimum": np.min, "multiply": np.prod}
    expected = functions[operation](expression_values, axis=-1).astype(NP_TYPES[dtype])
    if dtype == "i64":
        outputs = {arg["name"] for arg in meta["args"] if arg.get("output")}
        def encode_writes(stmt):
            if isinstance(stmt, Write) and stmt.output in outputs:
                return [Let(name="low_mask", value=Literal(65535, "int"), dtype=I64),
                            assignment("low_bits", "bitwise_and", (stmt.values[0], "low_mask"), "i64"),
                            replace(stmt, values=("low_bits",))]
            nested = stmt.nested()
            if nested:
                stmt = stmt.with_bodies(tuple(tuple(result for child in children for result in encode_writes(child)) for children in nested))
            return [stmt]
        body = [result for stmt in body for result in encode_writes(stmt)]
        expected &= np.int64(65535)
    source = replace(source, kernel=replace(source.kernel, body=(replace(tile, axes=axes, block_threads=work, body=tuple(body)),)))
    meta.update(block=[work, 1, 1], grid=[3, 1, 1])
    for arg in meta["args"]:
        if arg.get("output"):
            arg["dtype"] = dtype
        else:
            arg.pop("seed", None)
            arg["data"] = values.ravel().tolist()
    output = next(arg["name"] for arg in meta["args"] if arg.get("output"))
    full_meta = {"id": "typed_reduction_backend", "kernels": [meta], "graph_inputs": ["x"], "graph_outputs": [output],
                 "rtol": 0, "atol": 0}
    return source, full_meta, {output: expected}


def classification(values, dtype):
    values = np.asarray(values)
    return np.where(np.isnan(values), 5, np.where(np.isinf(values), 3 + np.signbit(values),
                    np.where(values == 0, np.signbit(values), 2))).astype(np.int32)


def check_classification(meta, ptx, expected, dtype):
    descriptor = meta["kernels"][0]
    assembled = post(RUNNER, "/compile", {"ptx": ptx})
    if not assembled.get("ok"):
        return "runner_assembly_failure", assembled
    request = {"cubin_id": assembled["cubin_id"], "entry": descriptor["entry"],
        "grid": descriptor["grid"], "block": descriptor["block"], "smem_bytes": 0, "zero_outputs": [],
        "args": descriptor["args"], "warmup": 0, "iters": 1, "max_return": 4096}
    for attempt in range(12):
        response = post(RUNNER, "/run", request)
        if response.get("stage") != "busy":
            break
        time.sleep(3)
    if not response.get("ok"):
        return "runner_execution_failure", response
    name = meta["graph_outputs"][0]
    if response["outputs"][name].get("truncated"):
        return "truncated_output", {}
    words = np.asarray(response["outputs"][name]["values"], np.int32).view(np.uint32)
    if dtype == "f64":
        exponent, fraction = words & 0x7ff00000, words & 0xfffff
        # All NaNs generated here have a nonzero high-word payload.
        decoded = ((words.astype(np.uint64) << 32).view(np.float64))
    else:
        decoded = words.view(np.float32)
    want, got = classification(expected[name], dtype), classification(decoded, dtype)
    return ("pass" if np.array_equal(want, got) else "wrong_classification",
            {"expected_classes": want.tolist(), "got_classes": got.tolist(), "class_codes":
             {"positive_zero": 0, "negative_zero": 1, "finite_nonzero": 2, "positive_inf": 3, "negative_inf": 4, "nan": 5}})


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--runner", action="store_true")
    args = parser.parse_args()
    output = ROOT / "review/operations"
    output.mkdir(parents=True, exist_ok=True)
    cases = []
    for dtype in ("f16", "f32", "f64"):
        for operation in ("exp", "log", "exp2", "log2", "log10", "power"):
            if operation.startswith("exp"):
                limit = {"f16": 10, "f32": 80, "f64": 700}[dtype]
                x = np.linspace(-limit, limit, 257)
                x[128] = 0
            elif operation.startswith("log"):
                bounds = {"f16": (-24, 15), "f32": (-149, 127), "f64": (-1074, 1023)}[dtype]
                x = np.exp2(np.linspace(*bounds, 257))
                x[128] = 1 + np.finfo(NP_TYPES[dtype]).eps
            else:
                x = np.resize(np.array([-2., -1., -.5, .1, 1., 2., 10.]), 257)
            y = np.resize(np.array([-3., -2., -1., 0., 1., 2., 3.]), len(x)) if operation == "power" else None
            cases.append((f"typed_{operation}_{dtype}", dtype, False, vector_case(dtype, operation, x, y)))
        for operation in ("sqrt", "reciprocal", "exp", "log", "power", "maximum_nan", "minimum_nan"):
            x = np.array([-2., -1., -0., 0., 1., 2., 1000., -1000.])
            y = np.array([.5, 3., -3., 0., 0., -3., 2., 3.])
            cases.append((f"class_{operation}_{dtype}", dtype, True, vector_case(dtype, operation, x, y, True)))
    for dtype in TYPES:
        for operation in ("maximum", "minimum", "multiply"):
            for work in (32, 128, 256):
                cases.append((f"typed_reduce_{operation}_{dtype}_w{work}", dtype, False, reduction_case(dtype, operation, work)))
    for dtype in ("i32", "i64"):
        cases.append((f"typed_power_{dtype}", dtype, False,
                      vector_case(dtype, "power", [-3, -2, -1, 0, 1, 2, 3], [0, 1, 2, 3, 4, 5, 6])))
        bits = 32 if dtype == "i32" else 64
        cases.append((f"typed_power_wrap_{dtype}", dtype, False,
                      vector_case(dtype, "power", [2, 2, -2, -3, 0], [bits-1, bits, bits-1, 3, 0])))
    rows = []
    for name, dtype, classify, (source, meta, expected) in cases:
        row = {"case": name, "kind": "classification" if classify else "typed_backend", "status": "failure"}
        folder = ROOT / "tests/operation-backends" / name
        try:
            # Classification fixtures store original expected numeric values
            # separately; their kernel output is a 32-bit encoded word.
            if classify:
                meta["comparison"] = "classify encoded float word output against expected_numeric.npz; ignore NaN payload/sign"
            save_source(folder, source, meta, expected,
                        expected_filename="expected_numeric.npz" if classify else "expected.npz")
            compiled_source = load_kernel(folder / f"{source.kernel.name}.ir.json")
            ptx = emit_ptx(compiled_source)
            path = folder / f"{source.kernel.name}.ours.ptx"
            path.write_text(ptx)
            subprocess.run(["ptxas", "-arch=sm_89", str(path), "-o", str(path.with_suffix(".cubin"))],
                           check=True, capture_output=True, text=True, timeout=60)
            row["status"] = "assembled_not_executed"
            if args.runner:
                if classify:
                    row["status"], row["details"] = check_classification(meta, ptx, expected, dtype)
                else:
                    ir_reference = simulate(folder)
                    row["status"], row["details"] = check_runner(folder, meta, {source.kernel.name: ptx}, ir_reference)
                time.sleep(1)
        except (Exception, SystemExit) as exc:
            row["error"] = str(exc)
            if isinstance(exc, subprocess.CalledProcessError):
                row["error"] += "\n" + (exc.stderr or "")[-5000:]
        rows.append(row)
        (output / "typed-backends.json").write_text(json.dumps({"source_sha256": {
            name: hashlib.sha256((ROOT/name).read_bytes()).hexdigest() for name in ("compile.py", "ptx_math.py")},
            "runner": args.runner, "cases": rows}, indent=2)+"\n")
        print(f"{row['status']:24} {name}", flush=True)
        if "error" in row:
            print(row["error"][-800:], flush=True)
    return int(any(row["status"] not in ("pass", "assembled_not_executed")
                   or row.get("details", {}).get("comparisons", {}).get("kernel_ir", {}).get("status", "pass") != "pass"
                   for row in rows))


if __name__ == "__main__":
    raise SystemExit(main())
